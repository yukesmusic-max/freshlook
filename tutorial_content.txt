# Each slide starts with a line "## slide" -- slides are numbered

# Right after the "## slide" line you may add any of these control lines
# (all optional, one per line, in any order). Stop as soon as you write a
# line that ISN'T one of these -- everything from there to the next
# "## slide" is the slide's body text.

# Within the body: consecutive lines with no blank line between them are
# treated as one paragraph and re-wrapped to fit the screen automatically,
# so you don't need to hand-wrap your sentences. A blank line forces an
# actual line break on screen -- use it to control exactly where one
# thought ends and the next begins.
#
#   k2_label: <text>      what K2's button says (default: "back")
#   k3_label: <text>      what K3's button says (default: "proceed")
#   k2: <action>          what K2 actually does   (default: go back one slide)
#   k3: <action>          what K3 actually does   (default: go forward one slide)
#   no_k2: true           hide the K2 button on this slide entirely
#   align: left           left-align this slide's text instead of centering it
#   outline: true          show the live category/subcategory tree (no
#                          counts, no [empty] tags) below this slide's own
#                          text, scrollable with E3 -- the only slide kind
#                          that uses an encoder at all
#
# Available actions for k2:/k3::
#   next              -- advance to the next slide (no label change needed)
#   skip_next         -- advance two slides, skipping the one right after --
#                        use this for a "No" that declines an action whose
#                        very next slide is a "Done, ..." follow-up that
#                        only makes sense if the action actually ran
#   back              -- go to the previous slide
#   end               -- end the tutorial now (same as the K1 long-press)
#   disable_quicknav  -- turns off Quick Nav, then advances
#   alphabetize       -- alphabetizes your favorites, then advances
#   autosort          -- runs Auto Sort, shows its results, then advances
#   replace_home      -- moves freshlook + blanks to the top, then advances
